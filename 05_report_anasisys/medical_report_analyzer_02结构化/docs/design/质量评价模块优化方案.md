```python
class MultiLayerValidator:
    """多层次验证器"""
    
    def __init__(self):
        self.syntax_validator = SyntaxValidator()          # 语法层验证
        self.semantic_validator = SemanticValidator()      # 语义层验证
        self.medical_validator = MedicalValidator()        # 医学逻辑层验证
        self.consistency_validator = ConsistencyValidator() # 一致性层验证
        self.confidence_scorer = ConfidenceScorer()        # 置信度评分
    
    def validate(self, result: Dict[str, Any]) -> ValidationResult:
        """分层验证流程"""
        validation_result = ValidationResult()
        
        # 第一层：语法验证
        syntax_issues = self.syntax_validator.validate(result)
        validation_result.add_layer_result("syntax", syntax_issues)
        
        # 第二层：语义验证
        semantic_issues = self.semantic_validator.validate(result)
        validation_result.add_layer_result("semantic", semantic_issues)
        
        # 第三层：医学逻辑验证
        medical_issues = self.medical_validator.validate(result)
        validation_result.add_layer_result("medical", medical_issues)
        
        # 第四层：一致性验证
        consistency_issues = self.consistency_validator.validate(result)
        validation_result.add_layer_result("consistency", consistency_issues)
        
        # 计算综合置信度
        confidence = self.confidence_scorer.calculate(validation_result)
        validation_result.set_confidence(confidence)
        
        return validation_result
```

```python
class ConfidenceScorer:
    """置信度评分器"""
    
    def __init__(self):
        self.weights = {
            'completeness': 0.25,    # 完整性权重
            'accuracy': 0.30,        # 准确性权重
            'consistency': 0.25,     # 一致性权重
            'medical_validity': 0.20  # 医学有效性权重
        }
    
    def calculate_confidence(self, validation_result: ValidationResult) -> float:
        """计算综合置信度"""
        scores = {
            'completeness': self._assess_completeness(validation_result),
            'accuracy': self._assess_accuracy(validation_result),
            'consistency': self._assess_consistency(validation_result),
            'medical_validity': self._assess_medical_validity(validation_result)
        }
        
        # 加权平均
        weighted_score = sum(
            scores[metric] * self.weights[metric] 
            for metric in scores
        )
        
        return min(max(weighted_score, 0.0), 1.0)
    
    def get_quality_grade(self, confidence: float) -> str:
        """获取质量等级"""
        if confidence >= 0.9:
            return "优秀"
        elif confidence >= 0.8:
            return "良好"
        elif confidence >= 0.7:
            return "合格"
        elif confidence >= 0.6:
            return "需要改进"
        else:
            return "质量较差"
    
    def get_improvement_suggestions(self, validation_result: ValidationResult) -> List[str]:
        """获取改进建议"""
        suggestions = []
        
        # 基于各维度得分提供具体建议
        if validation_result.completeness_score < 0.7:
            suggestions.append("建议补充缺失的病变特征信息")
        
        if validation_result.accuracy_score < 0.7:
            suggestions.append("建议检查解剖结构和诊断分类的准确性")
        
        if validation_result.consistency_score < 0.7:
            suggestions.append("建议检查影像发现与诊断的映射一致性")
        
        return suggestions
```

```python
class StatisticalAnomalyDetector:
    """统计异常检测器"""
    
    def detect(self, result: Dict, historical_data: List) -> List[Anomaly]:
        anomalies = []
        
        if not historical_data:
            return anomalies
        
        # 特征数量异常检测
        feature_count = len(result.get('结构化数据', {}).get('病变特征', []))
        historical_counts = [len(data.get('结构化数据', {}).get('病变特征', [])) 
                           for data in historical_data]
        
        if self._is_outlier(feature_count, historical_counts):
            anomalies.append(Anomaly(
                type="statistical",
                description=f"病变特征数量异常: {feature_count}个，历史平均: {np.mean(historical_counts):.1f}个",
                severity="warning"
            ))
        
        return anomalies
    
    def _is_outlier(self, value: float, data: List[float], threshold: float = 2.0) -> bool:
        """基于Z-score的异常值检测"""
        if len(data) < 3:
            return False
        
        mean = np.mean(data)
        std = np.std(data)
        
        if std == 0:
            return False
        
        z_score = abs((value - mean) / std)
        return z_score > threshold
```

```python
import asyncio
from concurrent.futures import ThreadPoolExecutor, ProcessPoolExecutor

class ParallelValidator:
    """并行验证器"""
    
    def __init__(self, max_workers: int = 4):
        self.thread_executor = ThreadPoolExecutor(max_workers=max_workers)
        self.process_executor = ProcessPoolExecutor(max_workers=max_workers)
    
    async def validate_batch(self, results: List[Dict]) -> List[ValidationResult]:
        """批量并行验证"""
        # 对于I/O密集型任务使用线程池
        tasks = []
        for result in results:
            task = asyncio.create_task(self._validate_single_async(result))
            tasks.append(task)
        
        return await asyncio.gather(*tasks)
    
    async def _validate_single_async(self, result: Dict) -> ValidationResult:
        """异步单个验证"""
        loop = asyncio.get_event_loop()
        
        # 在线程池中执行验证
        validation_result = await loop.run_in_executor(
            self.thread_executor,
            self._validate_single,
            result
        )
        
        return validation_result
    
    def validate_batch_parallel(self, results: List[Dict]) -> List[ValidationResult]:
        """使用进程池进行CPU密集型批量验证"""
        with self.process_executor as executor:
            futures = [executor.submit(self._validate_single, result) for result in results]
            return [future.result() for future in futures]
```

```python
class FeedbackLearner:
    """反馈学习系统"""
    
    def __init__(self):
        self.correction_analyzer = CorrectionAnalyzer()
        self.rule_generator = RuleGenerator()
        self.performance_tracker = PerformanceTracker()
    
    def learn_from_feedback(self, 
                          original_result: Dict, 
                          corrected_result: Dict, 
                          feedback: Dict) -> LearningResult:
        """从反馈中学习"""
        # 1. 分析修正模式
        correction_patterns = self.correction_analyzer.analyze(
            original_result, corrected_result
        )
        
        # 2. 生成新规则
        new_rules = self.rule_generator.generate_from_patterns(correction_patterns)
        
        # 3. 更新规则权重
        weight_updates = self._calculate_weight_updates(correction_patterns, feedback)
        
        # 4. 记录性能指标
        self.performance_tracker.record_feedback(feedback)
        
        return LearningResult(
            new_rules=new_rules,
            weight_updates=weight_updates,
            patterns=correction_patterns
        )
    
    def _calculate_weight_updates(self, patterns: List, feedback: Dict) -> Dict:
        """计算规则权重更新"""
        weight_updates = {}
        
        # 基于反馈质量调整权重
        feedback_quality = feedback.get('quality_score', 0.5)
        
        for pattern in patterns:
            rule_id = pattern.rule_id
            if pattern.correction_type == 'false_positive':
                # 误报，降低权重
                weight_updates[rule_id] = -0.1 * feedback_quality
            elif pattern.correction_type == 'false_negative':
                # 漏报，提高权重
                weight_updates[rule_id] = 0.1 * feedback_quality
        
        return weight_updates
```

```python
class RuleOptimizer:
    """规则优化引擎"""
    
    def __init__(self):
        self.performance_analyzer = PerformanceAnalyzer()
        self.rule_merger = RuleMerger()
        self.rule_pruner = RulePruner()
    
    def optimize_rules(self, performance_data: List[Dict]) -> OptimizationResult:
        """优化规则集"""
        # 1. 分析规则性能
        rule_performance = self.performance_analyzer.analyze(performance_data)
        
        # 2. 识别低效规则
        low_performers = self._identify_low_performers(rule_performance)
        
        # 3. 优化低效规则
        optimized_rules = self._optimize_low_performers(low_performers)
        
        # 4. 合并重复规则
        merged_rules = self.rule_merger.merge_redundant_rules(optimized_rules)
        
        # 5. 剪枝无效规则
        pruned_rules = self.rule_pruner.prune_ineffective_rules(merged_rules)
        
        return OptimizationResult(
            optimized_rules=pruned_rules,
            performance_improvement=self._calculate_improvement(rule_performance)
        )
    
    def _identify_low_performers(self, performance_data: Dict) -> List[str]:
        """识别低效规则"""
        low_performers = []
        
        for rule_id, metrics in performance_data.items():
            precision = metrics.get('precision', 0)
            recall = metrics.get('recall', 0)
            f1_score = metrics.get('f1_score', 0)
            
            # 设定性能阈值
            if f1_score < 0.6 or precision < 0.7:
                low_performers.append(rule_id)
        
        return low_performers
```

```python
class QualityMetrics:
    """质量指标计算器"""
    
    def calculate_comprehensive_metrics(self, result: Dict, ground_truth: Dict = None) -> Dict:
        """计算综合质量指标"""
        metrics = {
            # 基础指标
            'completeness_score': self._calc_completeness(result),
            'accuracy_score': self._calc_accuracy(result, ground_truth),
            'consistency_score': self._calc_consistency(result),
            'medical_validity_score': self._calc_medical_validity(result),
            'structure_quality_score': self._calc_structure_quality(result),
            
            # 高级指标

            'clinical_relevance_score': self._calc_clinical_relevance(result),
            'terminology_standardization_score': self._calc_terminology_standardization(result)
        }
        
        # 计算综合得分
        metrics['overall_score'] = self._calc_overall_score(metrics)
        
        return metrics
    
    def _calc_completeness(self, result: Dict) -> float:
        """计算完整性得分"""
        required_fields = {
            '解剖结构': 0.25,
            '病变特征': 0.35,
            '诊断信息': 0.25,
            '影像诊断映射': 0.15
        }
        
        score = 0.0
        structured_data = result.get('结构化数据', {})
        
        for field, weight in required_fields.items():
            field_data = structured_data.get(field, [])
            if field_data and len(field_data) > 0:
                # 基于数据丰富度计算得分
                richness = self._calc_field_richness(field_data)
                score += weight * richness
        
        return min(score, 1.0)
    
    def _calc_field_richness(self, field_data: List) -> float:
        """计算字段数据丰富度"""
        if not field_data:
            return 0.0
        
        total_richness = 0.0
        for item in field_data:
            item_richness = 0.0
            total_fields = 0
            filled_fields = 0
            
            if isinstance(item, dict):
                for key, value in item.items():
                    total_fields += 1
                    if value and str(value).strip():
                        filled_fields += 1
                
                if total_fields > 0:
                    item_richness = filled_fields / total_fields
            
            total_richness += item_richness
        
        return total_richness / len(field_data)
```

```python
class AdaptiveThresholds:
    """自适应阈值调整器"""
    
    def __init__(self):
        self.threshold_history = {}
        self.performance_history = []
    
    def adjust_thresholds(self, historical_data: List[Dict]) -> Dict[str, float]:
        """基于历史数据调整阈值"""
        new_thresholds = {}
        
        # 分析历史性能数据
        performance_stats = self._analyze_performance_stats(historical_data)
        
        # 调整各项阈值
        new_thresholds['completeness'] = self._optimize_threshold(
            performance_stats, 'completeness', target_precision=0.85
        )
        
        new_thresholds['accuracy'] = self._optimize_threshold(
            performance_stats, 'accuracy', target_precision=0.90
        )
        
        new_thresholds['consistency'] = self._optimize_threshold(
            performance_stats, 'consistency', target_precision=0.80
        )
        
        # 记录阈值变化
        self._record_threshold_changes(new_thresholds)
        
        return new_thresholds
    
    def _optimize_threshold(self, stats: Dict, metric: str, target_precision: float) -> float:
        """优化单个指标的阈值"""
        metric_values = stats.get(metric, [])
        if len(metric_values) < 10:  # 数据不足，使用默认值
            return 0.7
        
        # 使用ROC曲线分析找到最优阈值
        thresholds = np.linspace(0.1, 0.9, 50)
        best_threshold = 0.7
        best_f1 = 0.0
        
        for threshold in thresholds:
            precision, recall = self._calc_precision_recall(metric_values, threshold)
            f1 = 2 * (precision * recall) / (precision + recall) if (precision + recall) > 0 else 0
            
            if precision >= target_precision and f1 > best_f1:
                best_f1 = f1
                best_threshold = threshold
        
        return best_threshold
```
